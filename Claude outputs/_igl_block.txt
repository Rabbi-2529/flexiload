    // =========================================================================
    //  OFFER CHECK  (talks to the Django backend API at YOUR_SERVER_IP:8080)
    //  Auth is token-based: POST auth/login/ {email,password} -> {token},
    //  then send "Authorization: Token <token>" on every call.
    // =========================================================================

    private $iglBaseUrl = 'http://YOUR_SERVER_IP:8080/api/';
    private $iglEmail   = 'admin@example.com';
    private $iglPassword = 'YOUR_PASSWORD';
    private $iglToken   = null;

    // SIM Profile mapping (auto-populated from API)
    private $simProfiles = [];
    private $operatorToSimProfileId = [];

    // Operators shown in the form.
    private array $operatorChoices = [
        'grameen'  => 'Grameenphone',
        'rb'       => 'Robi',
        'at'       => 'Airtel',
        'bl'       => 'Banglalink',
        'teletalk' => 'Teletalk',
    ];

    // Filter buttons / category keys the offer cards use.
    private array $categoryLabels = [
        'all'       => 'All Offers',
        '30days'    => '30 Days',
        '7days'     => '7 Days',
        'data'      => 'Data',
        'minutes'   => 'Minutes',
        'combo'     => 'Combo',
        'social'    => 'Social',
        'streaming' => 'Streaming',
        'unlimited' => 'Unlimited',
    ];

    /** Log in to the backend and cache the token. Returns true on success. */
    private function authenticateIGL()
    {
        try {
            $client = new Client([
                'timeout' => 30,
                'verify'  => false,
            ]);

            $response = $client->post($this->iglBaseUrl . 'auth/login/', [
                'json' => [
                    'email'    => $this->iglEmail,
                    'password' => $this->iglPassword,
                ],
            ]);

            $body = json_decode($response->getBody(), true);

            if (isset($body['token'])) {
                $this->iglToken = $body['token'];
                Log::info('IGL Authentication successful');

                // Load SIM profiles after authentication.
                $this->loadSimProfiles();
                return true;
            }

            Log::error('IGL Authentication failed');
            return false;

        } catch (\Exception $e) {
            Log::error('IGL Authentication exception: ' . $e->getMessage());
            return false;
        }
    }

    /** Header array with the auth token. DRF uses the "Token" keyword. */
    private function iglHeaders()
    {
        return [
            'Authorization' => 'Token ' . $this->iglToken,
            'Accept'        => 'application/json',
        ];
    }

    /** Fetch SIM profiles and build operator -> sim_profile_id map. */
    private function loadSimProfiles()
    {
        try {
            $client = new Client([
                'timeout'     => 30,
                'verify'      => false,
                'http_errors' => false,
            ]);

            $response = $client->get($this->iglBaseUrl . 'sim-profiles/', [
                'headers' => $this->iglHeaders(),
            ]);

            if ($response->getStatusCode() < 200 || $response->getStatusCode() >= 300) {
                return;
            }

            $body = json_decode($response->getBody(), true);
            // The API may paginate: {results:[...]} or return a plain array.
            $list = $body['results'] ?? (is_array($body) ? $body : []);

            foreach ($list as $sim) {
                if (!isset($sim['operator'], $sim['id'])) {
                    continue;
                }
                $this->simProfiles[] = $sim;
                // Keep the first active SIM per operator.
                $op = $sim['operator']; // grameenphone / robi / airtel / banglalink
                if (!isset($this->operatorToSimProfileId[$op]) && ($sim['is_active'] ?? true)) {
                    $this->operatorToSimProfileId[$op] = $sim['id'];
                }
            }
        } catch (\Exception $e) {
            Log::error('IGL loadSimProfiles exception: ' . $e->getMessage());
        }
    }

    /** Show the empty form. */
    public function offerForm()
    {
        return view('user.flexiload.offer-check', [
            'operators'  => $this->operatorChoices,
            'categories' => $this->categoryLabels,
            'form'       => ['operator' => 'bl', 'number' => '', 'amount' => '10'],
            'result'     => null,
            'error'      => null,
        ]);
    }

    /** Handle the form submit and fetch offers from the backend. */
    public function checkOffer(Request $request)
    {
        $form = [
            'operator' => trim((string) $request->input('operator')),
            'number'   => trim((string) $request->input('number')),
            'amount'   => trim((string) $request->input('amount', '10')),
        ];

        $result = null;
        $error  = null;

        try {
            $result = $this->fetchOffers($form['operator'], $form['number'], $form['amount']);
        } catch (\Throwable $e) {
            $error = $e->getMessage();
        }

        return view('user.flexiload.offer-check', [
            'operators'  => $this->operatorChoices,
            'categories' => $this->categoryLabels,
            'form'       => $form,
            'result'     => $result,
            'error'      => $error,
        ]);
    }

    /** JSON endpoint (optional). */
    public function checkOfferApi(Request $request)
    {
        try {
            $result = $this->fetchOffers(
                (string) $request->input('operator'),
                (string) $request->input('number'),
                $request->input('amount', 10)
            );
        } catch (\Throwable $e) {
            return response()->json(['error' => $e->getMessage()], 400);
        }

        return response()->json([
            'operator' => $request->input('operator'),
            'number'   => $request->input('number'),
            'amount'   => $request->input('amount'),
            'count'    => $result['counts']['all'],
            'offers'   => $result['offers'],
        ]);
    }

    /**
     * Call the backend offer-check endpoint (token auth) and return the offers
     * plus per-category counts/filters the blade needs. The backend already
     * returns normalized offers with a 'cat' map, so no normalization here.
     *
     * @throws \InvalidArgumentException on bad input
     * @throws \RuntimeException on API/network problems
     */
    private function fetchOffers(string $operator, string $number, $amount): array
    {
        $operator = strtolower(trim($operator));
        $number   = trim($number);

        if (!preg_match('/^(?:\+?88)?01[3-9]\d{8}$/', $number)) {
            throw new \InvalidArgumentException('Enter a valid Bangladesh mobile number.');
        }
        $amount = (int) $amount;
        if ($amount <= 0) {
            throw new \InvalidArgumentException('Enter a valid amount.');
        }

        if (!$this->iglToken && !$this->authenticateIGL()) {
            throw new \RuntimeException('Could not sign in to the load server. Check the API credentials.');
        }

        try {
            $client = new Client([
                'timeout'     => (int) ($this->iglTimeout ?? 30),
                'verify'      => false,
                'http_errors' => false,
            ]);

            $payload = ['operator' => $operator, 'number' => $number, 'amount' => $amount];

            $response = $client->post($this->iglBaseUrl . 'offers/check/', [
                'headers' => $this->iglHeaders(),
                'json'    => $payload,
            ]);

            // Token expired / invalid -> re-authenticate once and retry.
            if ($response->getStatusCode() === 401) {
                if ($this->authenticateIGL()) {
                    $response = $client->post($this->iglBaseUrl . 'offers/check/', [
                        'headers' => $this->iglHeaders(),
                        'json'    => $payload,
                    ]);
                }
            }
        } catch (\Exception $e) {
            Log::error('Offer check request failed: ' . $e->getMessage());
            throw new \RuntimeException('Offer API request failed or timed out. Please try again.');
        }

        $status = $response->getStatusCode();
        $raw    = (string) $response->getBody();

        if ($status < 200 || $status >= 300) {
            Log::error("Offer check HTTP {$status}: " . substr($raw, 0, 300));
            // The API returns {"error":"..."} on 400 (bad number/operator).
            $decoded = json_decode($raw, true);
            if (isset($decoded['error'])) {
                throw new \RuntimeException($decoded['error']);
            }
            throw new \RuntimeException("Offer API error ({$status}). Please try again.");
        }

        $body   = json_decode($raw, true) ?: [];
        $offers = $body['offers'] ?? [];

        // Per-category counts for the filter buttons.
        $counts = array_fill_keys(array_keys($this->categoryLabels), 0);
        $counts['all'] = count($offers);
        foreach ($offers as $offer) {
            foreach (($offer['cat'] ?? []) as $key => $isOn) {
                if ($isOn && isset($counts[$key])) {
                    $counts[$key]++;
                }
            }
        }

        $filters = [];
        foreach ($this->categoryLabels as $key => $label) {
            $filters[] = ['key' => $key, 'label' => $label, 'count' => $counts[$key] ?? 0];
        }

        return [
            'offers'     => $offers,
            'counts'     => $counts,
            'filters'    => $filters,
            'message'    => $body['message'] ?? '',
            'uiOperator' => $operator,
        ];
    }

    /**
     * Buy an offer. This does NOT duplicate any load logic — it reshapes the
     * offer-check modal fields into the fields your existing single-load method
     * expects, then hands off to flexiloadFormProcess(). So LoadCamPending
     * insert, balance debit, etc. all stay in your one existing method.
     */
    public function buyOfferProcess(Request $request)
    {
        $request->validate([
            'offer_amount'    => 'required|integer|min:10|max:50000',
            'number_type'     => 'required',
            'targeted_number' => 'required',
            'flexipin'        => 'required',
            'operator'        => 'required',
        ]);

        // Map the offer-check operator (grameen/rb/at/bl) to the operator
        // string your load tables use (gp/robi/airtel/blink).
        $alias = [
            'grameen'  => 'gp',     'gp'     => 'gp',
            'rb'       => 'robi',   'robi'   => 'robi',
            'at'       => 'airtel', 'airtel' => 'airtel',
            'bl'       => 'blink',  'blink'  => 'blink',
            'teletalk' => 'teletalk',
        ];
        $operator = $alias[strtolower($request->operator)] ?? $request->operator;

        // Reshape into what flexiloadFormProcess() reads, then delegate to it.
        $request->merge([
            'amount'              => (int) $request->offer_amount,
            'operator'            => $operator,
            'dynamic_name_column' => $request->owner_name ?? '',
        ]);

        return $this->flexiloadFormProcess($request);
    }
